class SensorReading(Base):
    __tablename__ = "sensor_readings"

    id = Column(Integer, primary_key=True, index=True)
    humidity = Column(Float, nullable=False)
    soil_raw = Column(Integer, nullable=True)
    created_at = Column(
        DateTime(timezone=True),
        default=lambda: datetime.now(timezone.utc)
    )


# Crear las tablas
Base.metadata.create_all(bind=engine)

# Estado del servidor
@app.get("/api/status")
def status(db: Session = Depends(get_db)):
    last_reading = (
        db.query(SensorReading)
        .order_by(desc(SensorReading.id))
        .first()
    )

    online = False

    if last_reading:
        online = (
            datetime.now(timezone.utc)
            - last_reading.created_at.replace(tzinfo=timezone.utc)
        ).total_seconds() < 60

    return {
        "device_online": online,
        "last_humidity": (
            last_reading.humidity if last_reading else None
        )
    }


# Recibir mediciones del ESP32
@app.post("/api/readings")
def create_reading(
    humidity: float,
    soil_raw: int = None,
    db: Session = Depends(get_db)
):
    reading = SensorReading(
        humidity=humidity,
        soil_raw=soil_raw
    )

    db.add(reading)
    db.commit()
    db.refresh(reading)

    return {
        "message": "Medicion guardada correctamente",
        "id": reading.id,
        "humidity": reading.humidity
    }

@app.get("/api/readings")
def get_readings(db: Session = Depends(get_db)):
    readings = (
        db.query(SensorReading)
        .order_by(desc(SensorReading.id))
        .limit(20)
        .all()
    )

    return [
        {
            "id": r.id,
            "humidity": r.humidity,
            "soil_raw": r.soil_raw,
            "created_at": r.created_at
        }
        for r in readings
    ]

from sqlalchemy import create_engine
from sqlalchemy.orm import declarative_base, sessionmaker

DATABASE_URL = "sqlite:///./smart_irrigation.db"

engine = create_engine(
    DATABASE_URL,
    connect_args={"check_same_thread": False}
)

SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine
)

Base = declarative_base()


def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

const API = "";

let chart;

// Crear grÃ¡fica
function createChart() {
    const ctx = document.getElementById("humidityChart");

    chart = new Chart(ctx, {
        type: "line",

        data: {
            labels: [],
            datasets: [{
                label: "Humedad del suelo (%)",
                data: [],
                borderColor: "#198754",
                backgroundColor: "rgba(25,135,84,0.12)",
                fill: true,
                tension: 0.35
            }]
        },

        options: {
            responsive: true,
            scales: {
                y: {
                    min: 0,
                    max: 100
                }
            }
        }
    });
}

// Consultar estado
async function updateStatus() {
    try {
        const response = await fetch(`${API}/api/status`);

        if (!response.ok) throw new Error("API no disponible");

        const data = await response.json();

        document.getElementById("connectionStatus").textContent =
            "API conectada";

        document.getElementById("connectionDot").style.background =
            "#198754";

        document.getElementById("humidity").textContent =
            data.last_humidity !== null
                ? `${data.last_humidity}%`
                : "--%";

        document.getElementById("deviceStatus").textContent =
            data.device_online ? "En lÃ­nea" : "Sin datos recientes";

    } catch (error) {
        document.getElementById("connectionStatus").textContent =
            "API desconectada";

        document.getElementById("connectionDot").style.background =
            "#dc3545";
    }
}

// Consultar historial
async function updateHistory() {
    try {
        const response = await fetch(`${API}/api/readings`);

        if (!response.ok) throw new Error("Error al consultar");

        const readings = await response.json();

        const table = document.getElementById("historyTable");

        table.innerHTML = "";

        chart.data.labels = [];
        chart.data.datasets[0].data = [];

        readings.reverse().forEach(reading => {
            const date = new Date(reading.created_at);

            const row = document.createElement("tr");

            [
                reading.id,
                `${reading.humidity}%`,
                reading.soil_raw ?? "--",
                date.toLocaleString()
            ].forEach(value => {
                const cell = document.createElement("td");
                cell.textContent = value;
                row.appendChild(cell);
            });

            table.appendChild(row);

            chart.data.labels.push(
                date.toLocaleTimeString()
            );

            chart.data.datasets[0].data.push(
                reading.humidity
            );
        });

        chart.update();

    } catch (error) {
        console.error("Error en historial:", error);
    }
}

// Inicializar
document.addEventListener("DOMContentLoaded", () => {
    createChart();

    updateStatus();
    updateHistory();

    setInterval(updateStatus, 5000);
    setInterval(updateHistory, 10000);
});
